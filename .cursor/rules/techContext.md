## 项目结构
├── app.js # 应用入口
├── app.json # 应用配置
├── app.wxss # 全局样式
│   ├── calendar/ # 日历组件
│   ├── card/ # 卡片容器组件
│   ├── date-picker/ # 日期选择器
│   ├── float-menu/ # 浮动菜单
│   ├── progressBar/ # 进度条组件
│   ├── progressRing/ # 圆环进度组件
│   ├── recentTasks/ # 最近任务组件
│   ├── taskItem/ # 任务项组件
│   └── upcomingTask/ # 即将开始任务组件
├── pages/ # 页面文件
├── utils/ # 工具类
│   ├── dateUtils.js # 日期处理工具
│   ├── feedbackUtils.js # 反馈工具
│   ├── messageManager.js # 消息管理
│   ├── taskManager.js # 任务管理
│   ├── taskUtils.js # 任务工具
│   ├── uiUtils.js # UI工具
│   └── unit.js # 单位转换工具
└── assets/ # 静态资源
